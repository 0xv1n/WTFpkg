{{- $citations := .citations -}} {{- $references := .references -}} {{- if not $citations -}} {{- errorf "missing citations for a factual claim" -}} {{- else -}} {{- range $citation := $citations -}} {{- $number := int $citation -}} {{- if or (lt $number 1) (gt $number (len $references)) -}} {{- errorf "invalid citation %q: expected a reference number from 1 to %d" $citation (len $references) -}} {{- else -}} [{{ $number }}] {{- end -}} {{- end -}} {{- end -}}
